32 /*!****************************************************************************
33  * @file filterbank_feature_extract.h
34  * @brief Filterbank Feature Extraction — public API and types.
35  *
36  * This module implements filterbank-based feature extraction for keyword
37  * detection and similar machine-learning inference workloads. It provides
38  * an efficient pipeline that:
39  * - accepts a continuous stream of signed 16-bit PCM audio samples,
40  * - packs and overlaps frames into the format expected by the NPU,
41  * - runs MSB and LSB weight passes on the NPU hardware accelerator,
42  * - applies maxpool, scale, offset, and quantization post-processing, and
43  * - maintains a sliding feature window for the downstream classifier.
44  *
45  * Typical usage sequence:
46  * @code
47  * FBFE_Config cfg = { ... };
48  * FBFE_Ctx ctx;
49  * uint8_t features[FEATURES_BYTES];
50  *
51  * FBFE_Init(&ctx, &cfg, features, sizeof(features));
52  *
53  * while (audio_available())
54  * {
55  * FBFE_PushSample(&ctx, get_next_sample());
56  * }
57  *
58  * if (frame_complete())
59  * {
60  * FBFE_RunOneFrame(&ctx);
61  * // features[] now contains the updated sliding window
62  * }
63  * @endcode
64  *
65  ******************************************************************************/
